<script>
    function generatePass() {
        var flat = document.getElementById("flatNo").value.trim();
        var mobile = document.getElementById("mobileNo").value.trim();

        if (flat === "" || mobile === "") {
            alert("???? ????? ????? ???? ??? ?????? ???? ???? ???!");
            return;
        }

        document.getElementById("dispFlat").innerText = flat;
        document.getElementById("dispMob").innerText = mobile;
        document.getElementById("qrcode").innerHTML = "";

        var qrData = "VINAYAK|FLAT:" + flat + "|MOB:" + mobile;

        new QRCode(document.getElementById("qrcode"), {
            text: qrData,
            width: 180,
            height: 180,
            colorDark : "#000000",
            colorLight : "#ffffff",
            correctLevel : QRCode.CorrectLevel.H
        });

        document.getElementById("inputForm").style.display = "none";
        document.getElementById("passCard").style.display = "block";
    }
